local VectorCanvas = {}

function VectorCanvas.DrawSegment(parent, x1, y1, x2, y2, thickness, color, zIndex)
    local dx = x2 - x1
    local dy = y2 - y1
    local dist = math.sqrt(dx * dx + dy * dy)
    local cx = (x1 + x2) / 2
    local cy = (y1 + y2) / 2
    local angle = math.deg(math.atan2(dy, dx))

    local line = Instance.new("Frame")
    line.Name = "VectorSegment"
    line.BorderSizePixel = 0
    line.BackgroundColor3 = color or Color3.fromRGB(240, 244, 250)
    line.AnchorPoint = Vector2.new(0.5, 0.5)
    line.Position = UDim2.new(cx, 0, cy, 0)
    line.Size = UDim2.new(dist, 0, 0, thickness or 2)
    line.Rotation = angle
    line.ZIndex = zIndex or 5
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(1, 0)
    corner.Parent = line
    line.Parent = parent
    return line
end

function VectorCanvas.DrawCircleRing(parent, cx, cy, scaleRadius, strokeWidth, color, filled, zIndex)
    local ring = Instance.new("Frame")
    ring.Name = "VectorRing"
    ring.AnchorPoint = Vector2.new(0.5, 0.5)
    ring.Position = UDim2.new(cx, 0, cy, 0)
    ring.Size = UDim2.new(scaleRadius * 2, 0, scaleRadius * 2, 0)
    ring.BackgroundColor3 = color or Color3.fromRGB(240, 244, 250)
    ring.BackgroundTransparency = filled and 0 or 1
    ring.BorderSizePixel = 0
    ring.ZIndex = zIndex or 5
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(1, 0)
    c.Parent = ring
    if not filled then
        local st = Instance.new("UIStroke")
        st.Color = color or Color3.fromRGB(240, 244, 250)
        st.Thickness = strokeWidth or 1.8
        st.Parent = ring
    end
    ring.Parent = parent
    return ring
end

-- // DOKUNULAN YERDE İZ ÇIKMASI (TOUCH / CLICK RIPPLE TRAIL)
local function SpawnTouchRipple(x, y)
    if not OrbitConfig.RippleEnabled then return end
    local ripple = Instance.new("Frame")
    ripple.Name = "OrbitTouchRipple"
    ripple.AnchorPoint = Vector2.new(0.5, 0.5)
    ripple.Position = UDim2.new(0, x, 0, y)
    ripple.Size = UDim2.new(0, 8, 0, 8)
    ripple.BackgroundColor3 = Color3.fromRGB(56, 189, 248)
    ripple.BackgroundTransparency = 0.65
    ripple.BorderSizePixel = 0
    ripple.ZIndex = 999
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(1, 0)
    c.Parent = ripple
    local st = Instance.new("UIStroke")
    st.Color = Color3.fromRGB(186, 230, 253)
    st.Thickness = 1.8
    st.Transparency = 0.15
    st.Parent = ripple
    ripple.Parent = ScreenGui

    local tw = TweenService:Create(ripple, TweenInfo.new(0.48, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 68, 0, 68),
        BackgroundTransparency = 1
    })
    TweenService:Create(st, TweenInfo.new(0.48, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
        Transparency = 1
    }):Play()
    tw:Play()
    task.delay(0.5, function()
        if ripple and ripple.Parent then ripple:Destroy() end
    end)
end

-- // YAYLI BUTON EFEKTİ (Tıklayınca Küçülüp Bırakınca Anlık Büyüme + Ses)
local function AttachSpringEffect(btn, clickCallback, soundType)
    local scale = Instance.new("UIScale")
    scale.Scale = 1
    scale.Parent = btn

    btn.MouseButton1Down:Connect(function()
        PlaySound(soundType or "Click", 0.98)
        TweenService:Create(scale, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Scale = 0.92
        }):Play()
    end)

    btn.MouseButton1Up:Connect(function()
        TweenService:Create(scale, TweenInfo.new(0.16, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Scale = 1.04
        }):Play()
        task.delay(0.14, function()
            if scale and scale.Parent then
                TweenService:Create(scale, TweenInfo.new(0.12, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
                    Scale = 1
                }):Play()
            end
        end)
    end)

    btn.MouseLeave:Connect(function()
        TweenService:Create(scale, TweenInfo.new(0.12, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
            Scale = 1
        }):Play()
    end)

    if clickCallback then
        btn.MouseButton1Click:Connect(clickCallback)
    end
end

-- // P2P ANIMATE KANALI MİMARİSİ (Humanoid.Animator AnimationTrack Replikasyonu)
local P2PEngine = {}
P2PEngine.CarrierTrack = nil
P2PEngine.ShowBanner = nil

function P2PEngine.GetCarrierTrack()
    local char = LocalPlayer.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local animator = hum:FindFirstChildOfClass("Animator")
    if not animator then
        animator = Instance.new("Animator")
        animator.Parent = hum
    end

    if not P2PEngine.CarrierTrack or not P2PEngine.CarrierTrack.Parent then
        local anim = Instance.new("Animation")
        anim.AnimationId = OrbitConfig.CarrierAnimationId
        local ok, tr = pcall(function()
            return animator:LoadAnimation(anim)
        end)
        if ok and tr then
            tr.Priority = Enum.AnimationPriority.Action
            P2PEngine.CarrierTrack = tr
        end
    end
    return P2PEngine.CarrierTrack
end

-- OpCode'ları:
-- 1 = Global P2P Duyuru / Handshake
-- 2 = Bring (Hedefi Yanına Çek)
-- 3 = Freeze / Thaw (Hedefi Dondur / Çöz)
-- 4 = Fling (Hedefi Havaya Savur)
-- 5 = Görsel Aura Efekti (Orbit Halkası)
-- 6 = Kick (İstemci Oturum İzolasyonu)
-- 7 = Saat / Yerçekimi Ortam Senkronizasyonu
-- 8 = P2P Peer Discovery (Keşif Sinyali)
function P2PEngine.Broadcast(opCode, targetPlayer, extraValue)
    local track = P2PEngine.GetCarrierTrack()
    local targetHash = 0
    local targetName = "ALL"
    if targetPlayer and typeof(targetPlayer) == "Instance" and targetPlayer:IsA("Player") then
        targetHash = (targetPlayer.UserId % 80) + 10
        targetName = targetPlayer.Name
    else
        targetHash = 95 -- 95 = ALL
    end

    local encodedTimePos = opCode + (targetHash / 100)
    local encodedSpeed = 1.15 + ((opCode % 7) * 0.25)
    local encodedWeight = math.clamp((extraValue or 50) / 100, 0.1, 0.99)

    if track then
        pcall(function()
            track:Play(0.05, encodedWeight, encodedSpeed)
            track.TimePosition = encodedTimePos
            task.delay(0.35, function()
                if track and track.IsPlaying then
                    track:Stop(0.1)
                end
            end)
        end)
    end

    PlaySound("Broadcast", 1.04)
    AddLog(
        string.format("P2P_TX[0x0%d]", opCode),
        LocalPlayer.Name,
        string.format("Hedef: %s | TP: %.2f | Spd: %.2f | W: %.2f", targetName, encodedTimePos, encodedSpeed, encodedWeight)
    )
end

-- Diğer oyuncularda görsel vektörel halka (Aura) oluşturucu
local function ApplyPeerAura(player, enabled)
    if not player or not player.Character then return end
    local root = player.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local existing = root:FindFirstChild("OrbitP2PAura")
    if not enabled then
        if existing then existing:Destroy() end
        return
    end
    if existing then return end

    local bg = Instance.new("BillboardGui")
    bg.Name = "OrbitP2PAura"
    bg.Size = UDim2.new(6, 0, 6, 0)
    bg.AlwaysOnTop = false
    bg.Parent = root

    VectorCanvas.DrawCircleRing(bg, 0.5, 0.5, 0.42, 2.5, Color3.fromRGB(56, 189, 248), false, 3)
    VectorCanvas.DrawCircleRing(bg, 0.5, 0.5, 0.30, 1.5, Color3.fromRGB(168, 85, 247), false, 3)
end

-- Gelen P2P sinyallerini çöz ve uygula
local function HandleIncomingP2PFrame(sender, opCode, targetHash, weightVal)
    OrbitState.VerifiedPeers[sender.UserId] = true
    if RefreshPeersUI then pcall(RefreshPeersUI) end

    local myHash = (LocalPlayer.UserId % 80) + 10
    local isForMe = (targetHash == 95) or (math.abs(targetHash - myHash) <= 1)

    AddLog(
        string.format("P2P_RX[0x0%d]", opCode),
        sender.Name,
        string.format("Peer sinyali alındı (HedefHash: %d, W: %.2f)", targetHash, weightVal)
    )

    if opCode == 1 and P2PEngine.ShowBanner then
        P2PEngine.ShowBanner(sender.Name, "ORBIT P2P Senkron Sinyali & Sunucu Duyurusu!")
    elseif opCode == 2 and isForMe then
        -- Bring: Gönderen adminin yanına ışınlan
        if sender.Character and sender.Character:FindFirstChild("HumanoidRootPart") and LocalPlayer.Character then
            LocalPlayer.Character:PivotTo(sender.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -4))
        end
    elseif opCode == 3 and isForMe then
        -- Freeze / Thaw
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local r = LocalPlayer.Character.HumanoidRootPart
            r.Anchored = not r.Anchored
        end
    elseif opCode == 4 and isForMe then
        -- Fling
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            LocalPlayer.Character.HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(math.random(-180, 180), 280, math.random(-180, 180))
        end
    elseif opCode == 5 then
        ApplyPeerAura(sender, true)
    elseif opCode == 6 and isForMe and sender ~= LocalPlayer then
        if P2PEngine.ShowBanner then
            P2PEngine.ShowBanner(sender.Name, "P2P Moderasyon Uyarısı Aldınız!")
        end
    elseif opCode == 7 then
        Lighting.ClockTime = math.clamp( math.floor(weightVal * 24), 0, 24 )
    end
end

function P2PEngine.WatchPeer(plr)
    if plr == LocalPlayer then return end

    local function BindChar(char)
        local hum = char:WaitForChild("Humanoid", 8)
        if not hum then return end
        local animator = hum:WaitForChild("Animator", 8)
        if not animator then return end

        animator.AnimationPlayed:Connect(function(track)
            if not track or not track.Animation then return end
            local animId = tostring(track.Animation.AnimationId)
            if string.find(animId, "507766388") or animId == OrbitConfig.CarrierAnimationId then
                task.wait(0.04)
                local tp = track.TimePosition
                local opCode = math.floor(tp + 0.001)
                local targetHash = math.floor(((tp - opCode) * 100) + 0.5)
                if opCode >= 1 and opCode <= 8 then
                    HandleIncomingP2PFrame(plr, opCode, targetHash, track.WeightCurrent)
                end
            end
        end)
    end

    if plr.Character then task.spawn(BindChar, plr.Character) end
    plr.CharacterAdded:Connect(BindChar)

    -- Chat üzerinden [ORBIT] duyuru mesajlarını da yakala
    plr.Chatted:Connect(function(msg)
        if string.sub(msg, 1, 7) == "[ORBIT]" then
            OrbitState.VerifiedPeers[plr.UserId] = true
            local cleanText = string.sub(msg, 9)
            if P2PEngine.ShowBanner then
                P2PEngine.ShowBanner(plr.Name, cleanText)
            end
        end
    end)
end

local FlightLV = nil
local FlightAO = nil
local FlightAtt = nil

local function SetFlyEnabled(state)
    OrbitState.Flying = state
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    if not state then
        if FlightLV then FlightLV:Destroy(); FlightLV = nil end
        if FlightAO then FlightAO:Destroy(); FlightAO = nil end
        if FlightAtt then FlightAtt:Destroy(); FlightAtt = nil end
        hum.PlatformStand = false
        AddLog("HAREKET", LocalPlayer.Name, "Fly (Uçuş) kapatıldı.")
        return
    end

    hum.PlatformStand = true
    FlightAtt = Instance.new("Attachment")
    FlightAtt.Name = "OrbitFlyAtt"
    FlightAtt.Parent = root

    FlightLV = Instance.new("LinearVelocity")
    FlightLV.Attachment0 = FlightAtt
    FlightLV.MaxForce = math.huge
    FlightLV.VectorVelocity = Vector3.zero
    FlightLV.Parent = root

    FlightAO = Instance.new("AlignOrientation")
    FlightAO.Attachment0 = FlightAtt
    FlightAO.Mode = Enum.OrientationAlignmentMode.OneAttachment
    FlightAO.MaxTorque = math.huge
    FlightAO.Responsiveness = 50
    FlightAO.CFrame = Workspace.CurrentCamera.CFrame
    FlightAO.Parent = root

    P2PEngine.Broadcast(5, nil, 70)
    AddLog("HAREKET", LocalPlayer.Name, "Fly (Uçuş) aktif edildi. Hız: " .. tostring(OrbitConfig.FlySpeed))
end

    Instance.new("UICorner", CmdBox).CornerRadius = UDim.new(0, 10)

    local RunCmdBtn = Instance.new("TextButton")

    RunCmdBtn.ZIndex = 24
    RunCmdBtn.Parent = page
    Instance.new("UICorner", RunCmdBtn).CornerRadius = UDim.new(0, 10)

    ExecuteOrbitCommand = function(rawText)
        local clean = string.gsub(rawText or "", "^%s*(.-)%s*$", "%1")
        if clean == "" then return end
        if string.sub(clean, 1, 1) == OrbitConfig.Prefix or string.sub(clean, 1, 1) == ":" then
            clean = string.sub(clean, 2)
        end
        local parts = string.split(clean, " ")
        local cmd = string.lower(parts[1] or "")
        local arg1 = parts[2]

        if cmd == "fly" then
            if tonumber(arg1) then OrbitConfig.FlySpeed = tonumber(arg1) end
            SetFlyEnabled(true)
        elseif cmd == "unfly" then
            SetFlyEnabled(false)
        elseif cmd == "noclip" or cmd == "nc" then
            OrbitState.Noclip = not OrbitState.Noclip
            AddLog("KOMUT", LocalPlayer.Name, "Noclip: " .. tostring(OrbitState.Noclip))
        elseif cmd == "speed" or cmd == "ws" then
            local s = tonumber(arg1) or 60
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then
                LocalPlayer.Character:FindFirstChildOfClass("Humanoid").WalkSpeed = s
            end
            AddLog("KOMUT", LocalPlayer.Name, "WalkSpeed = " .. tostring(s))
        elseif cmd == "esp" then
            OrbitState.ESP = not OrbitState.ESP
            AddLog("KOMUT", LocalPlayer.Name, "Vektörel 2D ESP = " .. tostring(OrbitState.ESP))
        elseif cmd == "announce" or cmd == "duyuru" then
            table.remove(parts, 1)
            local msg = table.concat(parts, " ")
            P2PEngine.Broadcast(1, nil, 90)
            P2PEngine.ShowBanner(LocalPlayer.Name, msg ~= "" and msg or "ORBIT P2P Duyurusu!")
        end
    end

    AttachSpringEffect(RunCmdBtn, function()
        ExecuteOrbitCommand(CmdBox.Text)
        CmdBox.Text = ""
    end, "Click")

    LocalPlayer.Chatted:Connect(function(msg)
        if string.sub(msg, 1, 1) == OrbitConfig.Prefix then
            ExecuteOrbitCommand(msg)
        end
    end)

    local cmdHelp = {
        ":fly [hız] — Pürüzsüz CFrame + LinearVelocity uçuşu başlatır",
        ":unfly — Uçuş modunu kapatır",
        ":noclip — Duvarlardan geçme modunu açar/kapatır",
        ":speed <değer> — Karakter yürüme hızını değiştirir",
        ":esp — 2D Vektörel Oyuncu Kutu & Mesafe ESP katmanını açar",
        ":announce <mesaj> — P2P Animate kanalıyla tüm ORBIT kullanıcılarına duyuru geçer"
    }

    for i, hText in ipairs(cmdHelp) do
        local row = CreateSectionCard(page, UDim2.new(0, 22, 0, 72 + (i - 1) * 48), UDim2.new(1, -44, 0, 40))
        local l = Instance.new("TextLabel")
        l.BackgroundTransparency = 1
        l.Position = UDim2.new(0, 14, 0, 0)
        l.Size = UDim2.new(1, -28, 1, 0)
        l.Font = Enum.Font.Code
        l.TextSize = 12
        l.TextColor3 = Color3.fromRGB(226, 232, 240)
        l.TextXAlignment = Enum.TextXAlignment.Left
        l.Text = hText
        l.ZIndex = 24
        l.Parent = row
    end
end

-- // FİZİK, FLY, NOCLIP, ORBIT & VEKTÖREL 2D ESP ANA DÖNGÜSÜ
local EspBoxes = {}

RunService.RenderStepped:Connect(function(dt)
    local char = LocalPlayer.Character
    local cam = Workspace.CurrentCamera
    if char and cam then
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")

        -- 1. FLY (Hem PC WASD hem Mobil Thumbstick MoveDirection ile %100 Uyumlu)
        if OrbitState.Flying and root and hum and FlightLV and FlightAO then
            local moveDir = hum.MoveDirection
            local upDown = 0
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then upDown = upDown + 1 end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                upDown = upDown - 1
            end

            local camCF = cam.CFrame
            local targetVel = (moveDir * OrbitConfig.FlySpeed) + Vector3.new(0, (camCF.LookVector.Y * moveDir.Magnitude * OrbitConfig.FlySpeed) + (upDown * OrbitConfig.FlySpeed), 0)
            FlightLV.VectorVelocity = FlightLV.VectorVelocity:Lerp(targetVel, math.clamp(dt * 12, 0, 1))
            FlightAO.CFrame = camCF
        end

        -- 2. ORBIT (Seçili Oyuncu Etrafında Dönüş)
        if OrbitState.OrbitTarget and OrbitState.OrbitTarget.Character and root then
            local tRoot = OrbitState.OrbitTarget.Character:FindFirstChild("HumanoidRootPart")
            if tRoot then
                OrbitState.OrbitAngle = OrbitState.OrbitAngle + dt * 3.8
                local offset = Vector3.new(math.cos(OrbitState.OrbitAngle) * 8, 1.5, math.sin(OrbitState.OrbitAngle) * 8)
                root.CFrame = CFrame.new(tRoot.Position + offset, tRoot.Position)
            end
        end
    end

    -- 3. VEKTÖREL 2D ESP KATMANI
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local box = EspBoxes[plr]
            if OrbitState.ESP and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") and cam then
                local r = plr.Character.HumanoidRootPart
                local screenPos, onScreen = cam:WorldToViewportPoint(r.Position)
                if onScreen then
                    if not box then
                        box = Instance.new("Frame")
                        box.Name = "ESP_" .. plr.Name
                        box.AnchorPoint = Vector2.new(0.5, 0.5)
                        box.BackgroundTransparency = 1
                        box.Size = UDim2.new(0, 54, 0, 74)
                        box.ZIndex = 10
                        local st = Instance.new("UIStroke", box)
                        st.Color = OrbitState.VerifiedPeers[plr.UserId] and Color3.fromRGB(56, 189, 248) or Color3.fromRGB(34, 197, 94)
                        st.Thickness = 1.6
                        local nl = Instance.new("TextLabel", box)
                        nl.Name = "Info"
                        nl.BackgroundTransparency = 1
                        nl.Position = UDim2.new(0.5, -80, 0, -18)
                        nl.Size = UDim2.new(0, 160, 0, 16)
                        nl.Font = Enum.Font.Code
                        nl.TextSize = 11
                        nl.TextColor3 = Color3.fromRGB(56, 189, 248)
                        nl.Text = plr.Name
                        box.Parent = ScreenGui
                        EspBoxes[plr] = box
                    end
                    local dist = math.floor((cam.CFrame.Position - r.Position).Magnitude)
                    local scale = math.clamp(1200 / math.max(dist, 10), 28, 120)
                    box.Position = UDim2.new(0, screenPos.X, 0, screenPos.Y)
                    box.Size = UDim2.new(0, scale * 0.75, 0, scale)
                    box.Visible = true
                    if box:FindFirstChild("Info") then
                        box.Info.Text = string.format("%s [%dm]", plr.Name, dist)
                    end
                elseif box then
                    box.Visible = false
                end
            elseif box then
                box.Visible = false
            end
        end
    end
end)
